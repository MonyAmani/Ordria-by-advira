🛠️ إعداد قاعدة البيانات

في Supabase افتحي SQL Editor والصقي الكود ده مرة واحدة:

create table if not exists public.ordria_data (user_id uuid primary key references auth.users(id) on delete cascade, payload jsonb not null default '{}'::jsonb, updated_at timestamptz not null default now());
alter table public.ordria_data enable row level security;
drop policy if exists "ordria_select_own" on public.ordria_data;
drop policy if exists "ordria_insert_own" on public.ordria_data;
drop policy if exists "ordria_update_own" on public.ordria_data;
create policy "ordria_select_own" on public.ordria_data for select using (auth.uid() = user_id);
create policy "ordria_insert_own" on public.ordria_data for insert with check (auth.uid() = user_id);
create policy "ordria_update_own" on public.ordria_data for update using (auth.uid() = user_id) with check (auth.uid() = user_id);
⚠️ مفتاح Anon عادي يكون في واجهة الموقع، لكن ممنوع تحطي Service Role Key هنا.
